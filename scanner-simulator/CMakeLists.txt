cmake_minimum_required(VERSION 3.16)
project(microsight_scanner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(scanner_core src/scanner.cpp)
target_include_directories(scanner_core PUBLIC include)
add_executable(microsight_scanner src/main.cpp)
target_link_libraries(microsight_scanner PRIVATE scanner_core)
enable_testing()
add_executable(scanner_tests tests/scanner_tests.cpp)
target_link_libraries(scanner_tests PRIVATE scanner_core)
add_test(NAME scanner_tests COMMAND scanner_tests)
